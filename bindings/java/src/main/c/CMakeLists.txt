cmake_minimum_required(VERSION 3.20)
project(skp_jni C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# The core is built as a production static library: no test hooks, no sanitizers.
set(SKP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SKP_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(SKP_TESTING OFF CACHE BOOL "" FORCE)
set(SKP_SANITIZE OFF CACHE BOOL "" FORCE)
get_filename_component(SKP_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../core" ABSOLUTE)
add_subdirectory(${SKP_CORE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/core EXCLUDE_FROM_ALL)

if(NOT SKP_JAVA_HOME)
  set(SKP_JAVA_HOME $ENV{JAVA_HOME})
endif()
if(NOT SKP_JAVA_HOME)
  message(FATAL_ERROR "Pass -DSKP_JAVA_HOME=<jdk dir> (or set JAVA_HOME) so jni.h can be found")
endif()
set(SKP_JNI_INCLUDE "${SKP_JAVA_HOME}/include")
if(APPLE)
  list(APPEND SKP_JNI_INCLUDE "${SKP_JAVA_HOME}/include/darwin")
elseif(WIN32)
  list(APPEND SKP_JNI_INCLUDE "${SKP_JAVA_HOME}/include/win32")
else()
  list(APPEND SKP_JNI_INCLUDE "${SKP_JAVA_HOME}/include/linux")
endif()

add_library(skp_jni SHARED skp_jni.c)
target_include_directories(skp_jni PRIVATE ${SKP_JNI_INCLUDE} ${SODIUM_INCLUDE_DIRS})
target_compile_options(skp_jni PRIVATE -Wall -Wextra)
target_link_libraries(skp_jni PRIVATE skp_static)
set_target_properties(skp_jni PROPERTIES C_VISIBILITY_PRESET hidden)
