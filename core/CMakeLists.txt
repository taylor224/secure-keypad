cmake_minimum_required(VERSION 3.20)
project(skp VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(SKP_BUILD_TESTS "Build the test programs" ON)
option(SKP_TESTING "Compile in test hooks (injectable randomness). Never enable for production builds" OFF)
option(SKP_SANITIZE "Enable AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
option(SKP_BUILD_SHARED "Also build the shared library" ON)
set(SKP_SODIUM_STATIC_LIB "" CACHE FILEPATH "Path to libsodium.a to link statically (auto-detected when empty)")

if(SKP_BUILD_TESTS)
  set(SKP_TESTING ON CACHE BOOL "" FORCE)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED libsodium)
option(SKP_USE_ZLIB "Compress sprites with zlib when available (much smaller PNGs)" ON)
set(SKP_ZLIB_LINK)
if(SKP_USE_ZLIB)
  find_package(ZLIB)
  if(ZLIB_FOUND)
    set(SKP_ZLIB_LINK ZLIB::ZLIB)
    message(STATUS "zlib: ${ZLIB_VERSION_STRING}")
  else()
    message(STATUS "zlib: not found, using stb's compressor")
  endif()
endif()
if(SKP_SODIUM_STATIC_LIB STREQUAL "")
  find_library(SKP_SODIUM_STATIC_FOUND NAMES libsodium.a PATHS ${SODIUM_LIBRARY_DIRS} NO_DEFAULT_PATH)
  if(SKP_SODIUM_STATIC_FOUND)
    set(SKP_SODIUM_STATIC_LIB ${SKP_SODIUM_STATIC_FOUND})
  endif()
endif()
if(SKP_SODIUM_STATIC_LIB)
  message(STATUS "libsodium: static ${SKP_SODIUM_STATIC_LIB}")
  set(SKP_SODIUM_LINK ${SKP_SODIUM_STATIC_LIB})
else()
  message(STATUS "libsodium: shared (${SODIUM_LIBRARIES})")
  set(SKP_SODIUM_LINK ${SODIUM_LINK_LIBRARIES})
endif()

# ---- embedded fonts ------------------------------------------------------------------------
add_executable(skp_embed tools/embed.c)
set(SKP_FONT_SOURCES)
foreach(font Inter-Regular:skp_font_inter Roboto-Regular:skp_font_roboto)
  string(REPLACE ":" ";" parts ${font})
  list(GET parts 0 fname)
  list(GET parts 1 sym)
  set(src ${CMAKE_CURRENT_SOURCE_DIR}/fonts/${fname}.ttf)
  set(dst ${CMAKE_CURRENT_BINARY_DIR}/gen/${sym}.c)
  add_custom_command(OUTPUT ${dst}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/gen
    COMMAND skp_embed ${src} ${dst} ${sym}
    DEPENDS ${src} skp_embed
    COMMENT "Embedding ${fname}.ttf")
  list(APPEND SKP_FONT_SOURCES ${dst})
endforeach()

# ---- library ---------------------------------------------------------------------------------
set(SKP_SOURCES
  src/skp.c src/skp_crypto.c src/skp_layout.c src/skp_render.c src/skp_state.c src/skp_session.c
  third_party/cJSON.c ${SKP_FONT_SOURCES})

add_library(skp_objects OBJECT ${SKP_SOURCES})
target_include_directories(skp_objects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${SODIUM_INCLUDE_DIRS})
target_compile_options(skp_objects PRIVATE ${SODIUM_CFLAGS_OTHER} -Wall -Wextra -Wpedantic)
set_source_files_properties(third_party/cJSON.c ${SKP_FONT_SOURCES} PROPERTIES COMPILE_OPTIONS "-w")
if(SKP_TESTING)
  target_compile_definitions(skp_objects PUBLIC SKP_TESTING=1)
endif()
if(SKP_ZLIB_LINK)
  target_compile_definitions(skp_objects PRIVATE SKP_HAVE_ZLIB=1)
  target_link_libraries(skp_objects PRIVATE ZLIB::ZLIB)
endif()
if(SKP_SANITIZE)
  target_compile_options(skp_objects PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  set(SKP_SAN_LINK -fsanitize=address,undefined)
endif()

add_library(skp_static STATIC $<TARGET_OBJECTS:skp_objects>)
target_include_directories(skp_static PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(skp_static PUBLIC ${SKP_SODIUM_LINK} ${SKP_ZLIB_LINK} pthread m ${SKP_SAN_LINK})
set_target_properties(skp_static PROPERTIES OUTPUT_NAME skp)
if(SKP_TESTING)
  target_compile_definitions(skp_static INTERFACE SKP_TESTING=1)
endif()

if(SKP_BUILD_SHARED)
  add_library(skp SHARED $<TARGET_OBJECTS:skp_objects>)
  target_include_directories(skp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(skp PRIVATE ${SKP_SODIUM_LINK} ${SKP_ZLIB_LINK} pthread m ${SKP_SAN_LINK})
  target_compile_definitions(skp PRIVATE SKP_SHARED=1)
  set_target_properties(skp PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
endif()

# ---- keygen CLI -------------------------------------------------------------------------------
add_executable(skp-keygen tools/keygen.c)
target_link_libraries(skp-keygen PRIVATE skp_static)

# ---- tests -------------------------------------------------------------------------------------
if(SKP_BUILD_TESTS)
  enable_testing()
  foreach(t test_vectors test_render test_residue test_api)
    add_executable(${t} tests/${t}.c)
    target_link_libraries(${t} PRIVATE skp_static)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${SODIUM_INCLUDE_DIRS})
    target_compile_definitions(${t} PRIVATE SKP_VECTORS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../spec/vectors")
    add_test(NAME ${t} COMMAND ${t})
  endforeach()
  set_source_files_properties(tests/test_render.c PROPERTIES COMPILE_OPTIONS "-Wno-unused-function")
endif()

install(TARGETS skp_static ARCHIVE DESTINATION lib)
if(SKP_BUILD_SHARED)
  install(TARGETS skp LIBRARY DESTINATION lib)
endif()
install(FILES include/skp.h DESTINATION include)
